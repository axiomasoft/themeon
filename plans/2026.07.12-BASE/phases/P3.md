```ts
// src/types.ts
export interface StorageLike {
  getItem(key: string): string | null
  setItem(key: string, value: string): void
}
export interface UseThemeOptions {
  /** Известные имена тем; `toggle()` по умолчанию циклит первые две. Default `['light','dark']`. */
  themes?: readonly string[]
  /** Тема, когда нет сохранённой И не хотим системную. Если не задан — берётся системная. */
  default?: string
  /** Ключ localStorage; `null` отключает персист. Default `'themeon-theme'`. */
  storageKey?: string | null
  /** DOM-атрибут переключения. Default `'data-theme'` (D6). */
  attribute?: string
  /** Отображение системного предпочтения → имя темы. Default `{ dark:'dark', light:'light' }`. */
  system?: { dark: string; light: string }
  /** Глушить transition на кадр смены. Default `true`. */
  disableTransition?: boolean
  /** Runtime var-патчи для тем, которых НЕТ в статическом `tokens.css` (тенант/динамика, P6). */
  runtimeVars?: Readonly<Record<string, Record<string, string>>>
  /** Seam: целевой элемент. Default `() => document.documentElement`. */
  target?: () => import('@themeon/core').ElementLike & {
    setAttribute(n: string, v: string): void
  }
  /** Seam: хранилище. Default `() => localStorage`. */
  storage?: () => StorageLike | null
  /** Seam: медиа-квери. Default `(q) => matchMedia(q)`. */
  media?: (query: string) => { matches: boolean; addEventListener?: (t: 'change', cb: () => void) => void }
}
export interface UseThemeReturn {
  readonly theme: Readonly<Ref<string>>          // активная тема
  readonly system: Readonly<Ref<'dark' | 'light'>> // системное предпочтение
  readonly isDark: ComputedRef<boolean>          // theme === system.dark-имя
  set(theme: string): void
  toggle(a?: string, b?: string): void
  init(): void
}
```
- `useTheme(options?)`: получить/создать состояние (через `state.ts` — module-level singleton для plain-SPA ИЛИ инъекция из плагина в P3.2; в P3.1 — только module-level fallback + прямое создание). Вернуть `UseThemeReturn`.
- `toggle(a,b)`: `a ??= themes[0]; b ??= themes[1]; set(theme.value === a ? b : a)`.
- ЗАПРЕТЫ: не тянуть `@vueuse/core`; не писать в DOM на сервере; не создавать более одного источника активной темы; не хардкодить список тем внутри логики (только из опций/дефолта).
- Обязательные тесты (`use-theme.test.ts`, fake seams): (1) `init()` без stored + `prefers dark` → `theme='dark'`, атрибут выставлен; (2) `init()` с stored `'light'` при системной dark → `theme='light'` (persist побеждает); (3) `set('dark')` пишет в fake-storage и ставит атрибут; (4) `set` темы из `runtimeVars` вызывает `applyTheme` (проверить записанные `setProperty`), последующий `set` статической темы снимает те var (`removeProperty`); (5) `toggle()` без аргументов циклит `themes[0]↔themes[1]`; (6) SSR-нейтральность: `useTheme()` при `target()`-фейке без DOM не бросает при импорте/создании (запись — только в `set/init`); (7) `storageKey:null` не трогает storage; (8) `set` неизвестной темы (при заданном `themes`) → `console.warn`, но атрибут выставлен.

```ts
import { DEFAULTS } from './defaults' // только литералы, без vue/core
export interface ThemeInitScriptOptions {
  storageKey?: string
  attribute?: string
  darkTheme?: string
  lightTheme?: string
}
function assertSafeScriptToken(v: string, label: string): void {
  if (/['"`<>\\\n\r]/.test(v)) {
    throw new Error(`[themeon] themeInitScript: ${label} must not contain quotes, angle brackets, backslash or newlines: ${JSON.stringify(v)}`)
  }
}
export function themeInitScript(options: ThemeInitScriptOptions = {}): string {
  const k = options.storageKey ?? DEFAULTS.storageKey
  const a = options.attribute ?? DEFAULTS.attribute
  const d = options.darkTheme ?? DEFAULTS.darkTheme
  const l = options.lightTheme ?? DEFAULTS.lightTheme
  assertSafeScriptToken(k, 'storageKey'); assertSafeScriptToken(a, 'attribute')
  assertSafeScriptToken(d, 'darkTheme'); assertSafeScriptToken(l, 'lightTheme')
  return `(function(){try{var e=document.documentElement,s=localStorage.getItem('${k}'),t=s||(matchMedia('(prefers-color-scheme: dark)').matches?'${d}':'${l}');e.setAttribute('${a}',t)}catch(_){}})()`
}
```

```ts
import { createHash } from 'node:crypto'
import { readdirSync, readFileSync, statSync } from 'node:fs'
import { join, relative } from 'node:path'
export function hashDir(dir: string): string {
  const h = createHash('sha256')
  const walk = (d: string): void => {
    let entries: string[]
    try { entries = readdirSync(d).sort() } catch { return }
    for (const name of entries) {
      const p = join(d, name)
      const st = statSync(p)
      if (st.isDirectory()) walk(p)
      else { h.update(relative(dir, p)); h.update('\0'); h.update(readFileSync(p)) }
    }
  }
  walk(dir)
  return h.digest('hex')
}
```

**Implementation Rules:**
1. Опции: `{ theme: ThemeDefinition | (() => ThemeDefinition | Promise<ThemeDefinition>), tokensFiles?: string[], resolve?: ResolveOptions, serialize?: SerializeCssOptions, virtualId?: string (default 'virtual:themeon.css'), injectFouc?: boolean | ThemeInitScriptOptions }`. `theme` обязателен (без него — throw в фабрике плагина с ясным сообщением).
2. Virtual: `const V_ID = options.virtualId ?? 'virtual:themeon.css'`; `const RESOLVED = '\0' + V_ID`. `resolveId(id){ if (id === V_ID) return RESOLVED }`. `load(id){ if (id === RESOLVED) return await buildCss() }`. `buildCss()`: `const t = typeof theme === 'function' ? await theme() : theme; return serializeThemeCss(resolveTheme(t, options.resolve), options.serialize)`.
3. HMR `hotUpdate({ file, server })` (this-контекст даёт `this.environment`): если `tokensFiles` заданы и `file ∈ tokensFiles` (нормализовать пути) → `const mod = this.environment.moduleGraph.getModuleById(RESOLVED)`; если есть — `this.environment.moduleGraph.invalidateModule(mod)`; `this.environment.hot.send({ type:'update', updates:[{ type:'css-update', path: mod.url, acceptedPath: mod.url, timestamp: Date.now() }] })`; `return []` (обработали сами). Debounce ~10мс (батч частых сохранений — как UnoCSS). Если `tokensFiles` не заданы — HMR не активен (тема inline в конфиге, меняется через рестарт Vite-конфига).
4. `transformIndexHtml` (при `injectFouc`): вернуть tag-descriptor `{ tag:'script', children: themeInitScript(foucOpts), injectTo:'head-prepend' }` (Vite HTML-transform форма). `head-prepend` → скрипт до стилей/контента (анти-FOUC).
5. peer: `vite: '^7 || ^8'` (хук `hotUpdate` доступен с Vite 6, Environment API стабилен с 6 — диапазон безопасен, R-13 §4.2). `@themeon/core` — `dependencies`. `@themeon/vue` — `dependencies` (для `themeInitScript`), но `vue` — НЕ форсить: `@themeon/vue` объявляет `vue` peer; `@themeon/vite` не импортирует vue-рантайм (только pure `./anti-fouc`). Проверить, что установка `@themeon/vite` в проект без Vue не требует vue (тест: импорт плагина без vue в graph).
6. `sideEffects: false`. Русские комментарии на HMR-логике (Environment API, `\0`-конвенция, css-update vs js-update).
7. НЕ использовать `handleHotUpdate` (deprecated Vite 8, P-D26). НЕ использовать старые `server.moduleGraph`/`server.ws` (заменены `this.environment.*`).
**Code Guidance:**
- Эталон плагина (ядро):
```ts
import type { Plugin } from 'vite'
import { resolveTheme, serializeThemeCss, type ThemeDefinition } from '@themeon/core'
import { themeInitScript } from '@themeon/vue/anti-fouc'
export function themeon(options: ThemeonViteOptions): Plugin {
  if (!options?.theme) throw new Error('[themeon] vite plugin: `theme` option is required')
  const V_ID = options.virtualId ?? 'virtual:themeon.css'
  const RESOLVED = '\0' + V_ID
  const tokensFiles = new Set((options.tokensFiles ?? []).map((f) => normalizePath(f)))
  const buildCss = async (): Promise<string> => {
    const t = typeof options.theme === 'function' ? await options.theme() : options.theme
    return serializeThemeCss(resolveTheme(t, options.resolve), options.serialize)
  }
  return {
    name: 'themeon',
    resolveId(id) { if (id === V_ID) return RESOLVED },
    load(id) { if (id === RESOLVED) return buildCss() },
    hotUpdate({ file }) {
      if (!tokensFiles.has(normalizePath(file))) return
      const mod = this.environment.moduleGraph.getModuleById(RESOLVED)
      if (!mod) return
      this.environment.moduleGraph.invalidateModule(mod)
      this.environment.hot.send({ type: 'update', updates: [{
        type: 'css-update', path: mod.url, acceptedPath: mod.url, timestamp: Date.now(),
      }] })
      return []
    },
    transformIndexHtml() {
      if (!options.injectFouc) return
      const opts = options.injectFouc === true ? {} : options.injectFouc
      return [{ tag: 'script', children: themeInitScript(opts), injectTo: 'head-prepend' }]
    },
  }
}
```
  (debounce можно опустить в v1, если тесты стабильны — но заложить TODO-комментарий; UnoCSS дебонсит ~10мс. Решение исполнителя: если просто — без debounce, документировать.)
- ЗАПРЕТЫ: `handleHotUpdate`; `server.moduleGraph`/`server.ws` (устар.); импорт vue-рантайма; тема без `\0`-префикса resolved-id; хардкод пути токенов.
- Обязательные тесты (фейк-контекст): (1) `resolveId('virtual:themeon.css')` → `'\0virtual:themeon.css'`, чужой id → undefined; (2) `load(RESOLVED)` возвращает CSS, содержащий `@layer themeon.tokens` и `:root`; (3) `theme` как функция вызывается на каждый `load`; (4) `hotUpdate` для файла ∈ `tokensFiles` зовёт `invalidateModule` + `hot.send` с `css-update` и возвращает `[]` (фейк `this.environment`); (5) `hotUpdate` для чужого файла — no-op (не зовёт send); (6) `transformIndexHtml` при `injectFouc:true` возвращает script-tag `head-prepend` с IIFE; без опции — undefined; (7) фабрика без `theme` бросает.
- Тестирование `this`-контекста `hotUpdate`: вызвать `plugin.hotUpdate.call({ environment: fakeEnv }, { file, ... })`.

```ts
/**
 * Whether a value is a usable theme name. An empty (or whitespace-only) string is NOT a theme:
 * Nuxt coerces an undeclared `runtimeConfig` value to `''` (documented convention for
 * env-overridable keys), and a JSON/env transport can do the same. Treating `''` as a real theme
 * name kills the `prefers-color-scheme` fallback and poisons the persisted value.
 */
export function normalizeThemeName(value: unknown): string | undefined {
  return typeof value === 'string' && value.trim() !== '' ? value : undefined
}
```

смена канала доставки. Канал (`<link>`) сохраняется байт-в-байт → минимум FOUC/каскад-риска.

**Критическая находка детализатора (R-15 не знал):** модуль `@themeon/nuxt` (P3) в текущей
сборке **НЕ имеет опций `aliases`/`refLayer`** — его `setup()` зовёт `serializeThemeCss(

модуль-codegen НЕДОСТИЖИМЫ. Единственный путь эмитить `aliases: legacy-v0` — **CLI
`themeon build --aliases legacy-v0`** (`packages/cli/src/commands/build.ts:44-57` — флаги
`--aliases`/`--ref-layer` есть и валидируются). Отсюда архитектура P5 (P-D34): токены пилота
генерит `themeon build` в тот же `public/styles/base-vars.css`; `@themeon/nuxt` подключается с
`css: false` только ради анти-FOUC + auto-import `useTheme` + runtime-плагина, НЕ ради
theme-codegen. Модуль `@themeon/nuxt` в P5 **не правится**: пакеты — deliverable'ы своих фаз, пилот

  //   node themeon-parity.mjs <css> > snap.json                     — снять снимок
  //   node themeon-parity.mjs <css> --against baseline.json --expect '--size2xl,--size3xl'
  // Снимок: {"base":{"--x":"flat-val",...},"dark":{...}} — имена+flattened-значения, sorted.
  // ГЕЙТ падает (exit 1), если (removed ∪ changed) ⊄ expect. added — логируется (не роняет:
  //   канон-имена появляются рядом с легаси по D5 — это НЕ регресс, добавление var() ничего не
  //   ломает у потребителя, ломают ТОЛЬКО removed/changed).
  import { readFileSync } from 'node:fs'
  const norm = (v) => v.trim().replace(/\s+/g, ' ').replace(/#[0-9a-fA-F]{3,8}\b/g, (h) => h.toLowerCase())
  const themeOf = (sel) => {                       // селектор → ключ темы или null
    const s = sel.trim()
    if (s === ':root' || s.endsWith(' :root')) return 'base'
    const m = s.match(/\[data-theme=["']?([\w-]+)["']?\]/)
    return m ? m[1] : null
  }
  function parse(css) {                             // → { base:{--x:val}, dark:{...} } (raw declared)
    const noC = css.replace(/\/\*[\s\S]*?\*\//g, '')
    const blocks = {}, re = /([^{}]+)\{([^{}]*)\}/g; let m
    while ((m = re.exec(noC))) {
      const body = m[2]; if (!body.includes('--')) continue
      const t = themeOf(m[1]); if (t === null) continue
      const map = (blocks[t] ||= {})
      for (const d of body.split(';')) {
        const i = d.indexOf(':'); if (i < 0) continue
        const name = d.slice(0, i).trim(); if (!name.startsWith('--')) continue
        map[name] = norm(d.slice(i + 1))
      }
    }
    return blocks
  }
  function flatten(map) {                           // разворачивает var(--y[,fb]) до фикспоинта
    const out = {}
    for (const k of Object.keys(map)) {
      let v = map[k], n = 0
      while (/var\(\s*--/.test(v) && n++ < 12) {
        v = v.replace(/var\(\s*(--[\w-]+)\s*(?:,\s*([^()]*?))?\)/g, (_, ref, fb) =>
          map[ref] !== undefined ? map[ref] : (fb !== undefined ? fb.trim() : `var(${ref})`))
      }
      out[k] = norm(v)
    }
    return out
  }
  function effective(blocks) {                      // per-theme: base, base∪dark, … + flatten
    const base = blocks.base ?? {}, res = { base: flatten(base) }
    for (const t of Object.keys(blocks)) if (t !== 'base') res[t] = flatten({ ...base, ...blocks[t] })
    return res

    const o = base[t] ?? {}, n = sorted[t] ?? {}

    const added = Object.keys(n).filter((k) => !(k in o))

    if (added.length) console.error(`[added ${t}] ${added.join(', ')}`)
  }
  process.exit(fail ? 1 : 0)
